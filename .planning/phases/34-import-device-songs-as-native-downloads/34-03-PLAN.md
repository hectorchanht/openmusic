```typescript
export const IMPORT_RULES_KEY = 'openmusic:import-rules:v1';
export type PresetId = 'artist-title' | 'title-artist' | 'track-title';
export const PRESET_ORDER: readonly PresetId[];                 // ['artist-title', 'title-artist', 'track-title']
export const PRESET_LABELS: Record<PresetId, string>;           // LITERAL chip labels: '{artist} - {title}' | '{title} - {artist}' | '{track}. {title}'
export const IMPORT_EXTENSIONS: readonly string[];              // [...AUDIO_EXTENSIONS, 'opus']
export interface ImportRules {
	presets: PresetId[];        // multi-select, tried in PRESET_ORDER; first match wins
	stripTrackNo: boolean;
	stripBrackets: boolean;
	customPattern: string;      // '' = none; ONLY ever a value that passed validateCustomPattern
	minSeconds: number;         // 0..120, step 5; 0 = no length filter
	extensions: string[];       // subset of IMPORT_EXTENSIONS, lowercase, no dot
	skipRules: string[];        // one phrase per entry, matched case-insensitively against displayName AND relativePath
}
export const DEFAULT_IMPORT_RULES: ImportRules;                 // presets ['artist-title'], stripTrackNo true, stripBrackets false, customPattern '', minSeconds 30, extensions = AUDIO_EXTENSIONS (opus OFF), skipRules []
export function parseImportRules(raw: string | null): ImportRules;   // tolerant: unknown/missing → defaults per field, never throws
export interface ParsedName { title: string; artist: string; album?: string; track?: number }
export function stemOf(displayName: string): string;            // 'a/b.c.mp3' → 'b.c'  (no path, no last extension)
export function extOf(displayName: string): string;             // 'x.FLAC' → 'flac'; 'noext' → ''
export type PatternRejection = 'invalid' | 'no-groups' | 'too-slow';
export const PATTERN_PROBE_BUDGET_MS = 25;
export const PATTERN_PROBE_LEN = 24;
export function validateCustomPattern(src: string): { ok: true; re: RegExp } | { ok: false; reason: PatternRejection };
export function parseFilename(displayName: string, rules: ImportRules, custom: RegExp | null): ParsedName;  // D-16: never null; title falls back to the (stripped) stem
```
