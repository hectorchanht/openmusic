export type ImportPhase = 'idle' | 'requesting' | 'scanning' | 'done';
export type ImportNotice = { key: 'toast.importDone' | 'toast.importFailed' | 'toast.patternFellBack'; params?: Record<string, string | number> };
class DeviceImport {
	rules = $state<ImportRules>(DEFAULT_IMPORT_RULES);
	phase = $state<ImportPhase>('idle');
	done = $state(0);                         // rows checked so far — monotonic within a run
	total = $state(0);                        // 0 until the first page answers
	summary = $state<ImportSummary | null>(null);
	permission = $state<'denied' | 'denied-permanently' | null>(null);
	patternError = $state<PatternRejection | null>(null);
	notice = $state<ImportNotice | null>(null);   // store→UI channel; the (app) layout host maps via t() then clears
	load(): void;                             // idempotent, browser-guarded; hydrates rules from IMPORT_RULES_KEY

	setCustomPattern(src: string): boolean;   // '' clears; else validateCustomPattern → on ok persist + patternError=null; on reject patternError=reason, rules UNCHANGED; returns ok
	runImport(): Promise<void>;               // no-op when phase is 'requesting' | 'scanning'
	cancel(): void;                           // bumps importGen; loop bails after the current page; partial results kept; no drop pass
}
export const deviceImport: DeviceImport;
