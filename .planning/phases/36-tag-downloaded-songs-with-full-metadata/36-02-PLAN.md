```typescript
/** Everything we might know. An absent field is simply not written — never a placeholder (36-D-10). */
export interface AudioTagFields {
	title?: string;
	artist?: string;
	album?: string;        // 36-D-10: absent when Track.album is ''
	albumArtist?: string;  // 36-D-12: album-page artist, else the track's own artist
	trackNumber?: string;  // 36-D-11: ONLY from the album download loop
	// 36-D-07 no year/genre · 36-D-08 no comment/encoder · 36-D-09 no duration
}
export type AudioContainer = 'mp3' | 'm4a' | 'flac';
/** Result discriminant the CALLER logs (audio-tags.ts stays store-free — logAction is a runes store). */
export type TagOutcome =
	| { blob: Blob; result: 'tagged'; format: AudioContainer }
	| { blob: Blob; result: 'skipped-size' | 'unknown-container' | 'no-fields' | 'error' };
/** Memory ceiling. Measured peak RSS ≈ 6× file size (50.5 MB FLAC → 392 MB), Emscripten heap never shrinks. */
export const TAG_MAX_BYTES = 40 * 1024 * 1024;
export async function writeAudioTags(bytes: Uint8Array, fields: AudioTagFields, art?: { data: Uint8Array; mimeType: string } | null): Promise<{ bytes: Uint8Array; format: AudioContainer } | null>;
export async function readAudioTags(bytes: Uint8Array): Promise<(AudioTagFields & { format: AudioContainer }) | null>;
/** Blob-level wrapper used by download-track.ts and retag.ts. NEVER rejects; worst case returns the input blob. */
export async function tagAudioBlob(blob: Blob, fields: AudioTagFields, artDataUrl?: string | null): Promise<TagOutcome>;
/** `data:<mime>;base64,<b64>` → bytes + mime, or null. Inverse of media-artwork.ts bytesToBase64. */
export function dataUrlToBytes(dataUrl: string): { data: Uint8Array; mimeType: string } | null;
```
