```typescript
export function songKey(artistIn: string, titleIn: string): string
// returns `${strip(stripTranslation(title))}|${artist}` where strip() removes every char that is
// not \p{L} or \p{N}. Consequence: NEITHER half can contain '|', so every key has exactly ONE '|'.

function key(t: Track): string { return songKey(t.artist, t.title); }

export function sameSongKey(a: Track, b: Track): boolean {
	const ka = key(a);
	if (!ka || ka === '|') return false;
	return ka === key(b);
}
```
